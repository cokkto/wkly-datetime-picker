Simple to install

Install @wkly/datetime-picker with the same major version as your Angular application (11–22). For Angular 19, choose @wkly/datetime-picker@19. Read the documentation ↗

Import the picker module, then bind your selected value:

// booking.component.ts (standalone Angular component)
import { Component } from '@angular/core';
import {
  WklyDateTimePickerModule,
  WklyPickerValue,
} from '@wkly/datetime-picker';

@Component({
  selector: 'app-booking',
  standalone: true,
  imports: [WklyDateTimePickerModule],
  templateUrl: './booking.component.html',
})
export class BookingComponent {
  value: WklyPickerValue = null;
}
<!-- booking.component.html -->
<wkly-datetime-picker
  mode="datetime"
  [(value)]="value"
  ariaLabel="Booking date and time">
</wkly-datetime-picker>
